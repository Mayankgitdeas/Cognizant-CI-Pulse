# Python dependencies for Cognizant Competitor Pulse (admin-panel edition)
# Install with: pip install -r requirements.txt

fastapi          # Web framework
uvicorn[standard]  # ASGI server
sqlmodel          # Database ORM
pydantic-settings   # Environment variable management
python-dotenv    # Loads .env files
python-multipart  # Form submission support
itsdangerous       # Secure session cookies for admin login
bcrypt            # Password hashing
beautifulsoup4   # HTML parsing for the scraper
httpx            # Async HTTP client used by the scraper
